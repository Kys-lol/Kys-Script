    local function loadBypassButtonPosition()
        local pos = nil
        pcall(function()
            if isfile and readfile and isfile(DragConfigPath) then
                local raw = readfile(DragConfigPath)
                local data = HttpService:JSONDecode(raw)
                if data and data.XScale ~= nil and data.XOffset ~= nil and data.YScale ~= nil and data.YOffset ~= nil then
                    pos = UDim2.new(data.XScale, data.XOffset, data.YScale, data.YOffset)
                end
            end
        end)
        return pos
    end

    local function saveBypassButtonPosition(udim2Pos)
        pcall(function()
            if writefile then
                writefile(DragConfigPath, HttpService:JSONEncode({
                    XScale  = udim2Pos.X.Scale,
                    XOffset = udim2Pos.X.Offset,
                    YScale  = udim2Pos.Y.Scale,
                    YOffset = udim2Pos.Y.Offset,
                }))
            end
        end)
    end

    local function disconnectDragConns()
        for _, c in ipairs(bypassButtonDragConns) do
            pcall(function() c:Disconnect() end)
        end
        bypassButtonDragConns = {}
    end

    local function makeButtonDraggable(button)
        disconnectDragConns()
        local dragging = false
        local dragInput, dragStart, startPos

        local function update(input)
            local delta = input.Position - dragStart
            local newPos = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
            button.Position = newPos
        end

        table.insert(bypassButtonDragConns, button.InputBegan:Connect(function(input)
            if not getgenv().VD.SURV_DraggableGenBypass then return end
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                dragStart = input.Position
                startPos = button.Position
                local conn
                conn = input.Changed:Connect(function()
                    if input.UserInputState == Enum.UserInputState.End then
                        if dragging then
                            dragging = false
                            saveBypassButtonPosition(button.Position)
                        end
                        if conn then conn:Disconnect() end
                    end
                end)
            end
        end))

        table.insert(bypassButtonDragConns, button.InputChanged:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
                dragInput = input
            end
        end))

        table.insert(bypassButtonDragConns, UserInputService.InputChanged:Connect(function(input)
            if dragging and input == dragInput then
                update(input)
            end
        end))
    end

    -- ========== OPTIMIZED DoGenBoost ==========
    local function DoGenBoost()
        local char = LocalPlayer.Character
        if not char then return end
        local ci = char:FindFirstChild("CheckInterractable")
        if not ci then return end
        local repairing = ci:GetAttribute("isRepairing") or ci:GetAttribute("IsRepairing")
        if not repairing then return end

        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        -- Cache generator objects (refresh every 5 seconds)
        local genCache, lastCacheTime = {}, 0
        local function getGenerators()
            if tick() - lastCacheTime < 5 then return genCache end
            genCache, lastCacheTime = {}, tick()
            local folder = Workspace:FindFirstChild("Map") or Workspace
            for _, v in pairs(folder:GetDescendants()) do
                if v:IsA("Model") and v.Name == "Generator" then
                    local real = v:GetAttribute("RepairProgress") ~= nil
                        or v:GetAttribute("kickcount") ~= nil
                        or v:GetAttribute("ProgressRepair") ~= nil
                    if real then
                        table.insert(genCache, v)
                    end
                end
            end
            return genCache
        end

        local function getPoints(genModel)
            local pts = {}
            for _, obj in pairs(genModel:GetChildren()) do
                if obj.Name:find("GeneratorPoint") and obj:IsA("BasePart") then
                    table.insert(pts, obj)
                end
            end
            return pts
        end

        local function waitRepairing(point, timeout)
            local start = tick()
            while tick() - start < timeout do
                if point:GetAttribute("IsRepairing") == true then
                    return true
                end
                task.wait(0.05)
            end
            return false
        end

        local RepairEvent = nil
        pcall(function()
            local remotes = ReplicatedStorage:FindFirstChild("Remotes")
            if remotes then
                local genFolder = remotes:FindFirstChild("Generator")
                if genFolder then
                    RepairEvent = genFolder:FindFirstChild("RepairEvent")
                end
            end
        end)
        if not RepairEvent then return end

        local bestPoint, bestDist = nil, math.huge
        local bestGen = nil
        for _, gen in pairs(getGenerators()) do
            for _, pt in pairs(getPoints(gen)) do
                local d = (hrp.Position - pt.Position).Magnitude
                if d < bestDist then
                    bestDist = d
                    bestPoint = pt
                    bestGen = gen
                end
            end
        end

        if bestPoint and bestGen then
            local allPoints = getPoints(bestGen)
            local targetPoints = {}
            for _, p in ipairs(allPoints) do
                if p ~= bestPoint then table.insert(targetPoints, p) end
            end
            if #targetPoints == 0 then return end

            -- Batasi jumlah teleport untuk mengurangi lag (maks 5 titik)
            local maxTeleports = 5
            local teleported = 0
            local startCFrame = hrp.CFrame

            for i, point in ipairs(targetPoints) do
                if teleported >= maxTeleports then break end
                if not point.Parent then continue end

                hrp.Anchored = true
                hrp.CFrame = point.CFrame
                task.wait(0.05) -- lebih cepat dari 0.15

                if RepairEvent then
                    RepairEvent:FireServer(point, true)
                end

                -- Timeout lebih pendek (0.3 detik)
                local ok = waitRepairing(point, 0.3)
                if not ok then
                    if RepairEvent then RepairEvent:FireServer(point, false) end
                    task.wait(0.05)
                    hrp.CFrame = point.CFrame
                    task.wait(0.05)
                    if RepairEvent then RepairEvent:FireServer(point, true) end
                    waitRepairing(point, 0.2)
                end

                hrp.Anchored = false
                task.wait(0.02)

                teleported = teleported + 1
                -- Jeda antar titik lebih singkat (0.2 detik)
                task.wait(0.2)
            end

            pcall(function()
                hrp.Anchored = false
                hrp.CFrame = startCFrame
            end)
            local lastPoint = targetPoints[#targetPoints]
            if lastPoint and RepairEvent then
                task.wait(0.05)
                RepairEvent:FireServer(lastPoint, false)
            end
        end
    end

            if not bypassButton then return end
            local char = LocalPlayer.Character
            if not char then
                bypassButton.ImageColor3 = Color3.new(1, 1, 1)
                return
            end
            local ci = char:FindFirstChild("CheckInterractable")
            if not ci then
                bypassButton.ImageColor3 = Color3.new(1, 1, 1)
                return
            end
            local repairing = ci:GetAttribute("isRepairing") or ci:GetAttribute("IsRepairing")
            bypassButton.ImageColor3 = repairing and Color3.fromRGB(255, 140, 0) or Color3.new(1, 1, 1)
        end

            if not character then return end
            local ci = character:WaitForChild("CheckInterractable")
            if ci then
                if bypassButtonCheck then bypassButtonCheck:Disconnect() end
                bypassButtonCheck = ci:GetAttributeChangedSignal("isRepairing"):Connect(updateButtonColor)
                ci:GetAttributeChangedSignal("IsRepairing"):Connect(updateButtonColor)
                updateButtonColor()
            end
        end

        if LocalPlayer.Character then bindCheck(LocalPlayer.Character) end
        LocalPlayer.CharacterAdded:Connect(bindCheck)

        bypassButton.MouseButton1Click:Connect(function()
            if getgenv().VD.SURV_DraggableGenBypass then return end
            DoGenBoost()
        end)
    end

            bypassButtonGui:Destroy()
            bypassButtonGui = nil

        bypassButton = nil
        if bypassButtonCheck then
            bypassButtonCheck:Disconnect()
            bypassButtonCheck = nil

        end

    function startGenBoost()
        if not getgenv().VD.SURV_GenBoost then return end
        createBypassButton()
        bypassGuardianActive = true

        if keybindConnection then
            keybindConnection:Disconnect()
        end
        keybindConnection = UserInputService.InputBegan:Connect(function(input, gameProcessed)
            if gameProcessed then return end
            if input.KeyCode == Enum.KeyCode.G then
                DoGenBoost()
            end
        end)

        task.spawn(function()
            while bypassGuardianActive and getgenv().VD.SURV_GenBoost do
                if not (bypassButtonGui and bypassButtonGui.Parent) then
                    pcall(createBypassButton)
                end
                task.wait(1)
            end
        end)
    end

    function stopGenBoost()
        destroyBypassButton()
    end

    local TeamColor  = Color3.fromRGB(0, 255, 0)

    local function GetRole()
        if not LocalPlayer.Team then return "Lobby" end
        local name = LocalPlayer.Team.Name
        if name == "Killer" then return "Killer" end
        if name == "Survivors" then return "Survivor" end
        if name == "Spectator" then return "Spectator" end
        return "Lobby"
    end

    local function IsKiller(player)
        return player and player.Team and player.Team.Name == "Killer"
    end

    local function IsSurvivor(player)
        return player and player.Team and player.Team.Name == "Survivors"
    end

    local function IsSpectator(player)
        return player and player.Team and player.Team.Name == "Spectator"
    end

    local function isTeammate(player)
        if player == LocalPlayer then return false end
        local myRole = GetRole()
        if myRole == "Killer" then return IsKiller(player) end
        if myRole == "Survivor" then return IsSurvivor(player) end
        if myRole == "Spectator" then return IsSpectator(player) end
        return false
    end

    local function getPlayerColor(player)
        if IsSpectator(player) then return SpectatorColor end
        return isTeammate(player) and TeamColor or EnemyColor
    end

            if child.Name == itemName and (child:IsA("Model") or child:IsA("Tool") or child:IsA("Accessory")) then

        return false
    end

    local _fpWasSet = false

    local function RestoreFirstPersonCamera()
        if not _fpWasSet then return end
        _fpWasSet = false

        pcall(function()
            if _fpOriginal then
                LocalPlayer.CameraMode = _fpOriginal.CameraMode or Enum.CameraMode.Classic
                LocalPlayer.CameraMaxZoomDistance = _fpOriginal.CameraMaxZoomDistance or 128
                LocalPlayer.CameraMinZoomDistance = _fpOriginal.CameraMinZoomDistance or 0.5
            else
                LocalPlayer.CameraMode = Enum.CameraMode.Classic
                LocalPlayer.CameraMaxZoomDistance = 128
            end
        end)

        local char = LocalPlayer.Character
        if char then
            local head = char:FindFirstChild("Head")
            if head then head.LocalTransparencyModifier = 0 end
            for _, obj in ipairs(char:GetChildren()) do
                if obj:IsA("Accessory") then
                    local handle = obj:FindFirstChild("Handle")
                    if handle then handle.LocalTransparencyModifier = 0 end
                end
            end
        end

        _fpOriginal = nil
    end

    local function FaceLookDirection()
        local camera = Workspace.CurrentCamera
        if not camera then return end
        local rootPart = Root
        if not rootPart or not rootPart.Parent then return end

        local lookVec = camera.CFrame.LookVector
        local flat = Vector3.new(lookVec.X, 0, lookVec.Z)
        if flat.Magnitude <= 0 then return end

        local targetCF = CFrame.new(rootPart.Position, rootPart.Position + flat.Unit)
        local tween = TweenService:Create(rootPart, TweenInfo.new(0.2, Enum.EasingStyle.Linear), { CFrame = targetCF })
        tween:Play()
        tween.Completed:Connect(function()
            if Humanoid then Humanoid.AutoRotate = true end
            if ParrySystem.LockConnection then ParrySystem.LockConnection:Disconnect() end
            local startTime = tick()
            ParrySystem.LockConnection = RunService.Heartbeat:Connect(function()
                if tick() - startTime >= 0.8 then
                    if ParrySystem.LockConnection then
                        ParrySystem.LockConnection:Disconnect()
                        ParrySystem.LockConnection = nil
                    end
                    return
                end
                if rootPart and rootPart.Parent then
                    rootPart.CFrame = CFrame.new(rootPart.Position, rootPart.Position + flat.Unit)
                else
                    if ParrySystem.LockConnection then
                        ParrySystem.LockConnection:Disconnect()
                        ParrySystem.LockConnection = nil
                    end
                end
            end)
        end)
    end

    local function AttachParrySensor(kChar)

        local humanoid = kChar:FindFirstChild("Humanoid")
        if not humanoid then

            if not parent then
                Attached[kChar] = nil
            end
        end)

            local animId = track.Animation and track.Animation.AnimationId or ""
            if not VD_ATTACK_ANIMS[animId] then return end

            if GetRole() ~= "Survivor" then return end
            if not HasParryingDagger() then return end
            if not VD.SURV_AutoParry then return end
            if ParrySystem.IsOnCooldown or ParrySystem.IsResolving then return end

            local myChar = LocalPlayer.Character
            if not myChar then return end
            local myHRP = myChar:FindFirstChild("HumanoidRootPart")
            local kHRP = kChar:FindFirstChild("HumanoidRootPart")
            if not myHRP or not kHRP then return end

            local delta = myHRP.Position - kHRP.Position
            local startDistance = delta.Magnitude

            local mode = VD.SURV_ParryMode or "Legit"
            local range = VD.SURV_ParryRange or 12

            if mode == "Aggressive" then
                local aggressiveRadius = range
                local detectionRadius = aggressiveRadius + 3
                if startDistance > detectionRadius then return end
                if startDistance <= aggressiveRadius then
                    DoParry()
                else
                    local tracker
                    local startTime = os.clock()
                    tracker = RunService.Heartbeat:Connect(function()
                        if os.clock() - startTime >= 1.5 or ParrySystem.IsOnCooldown or ParrySystem.IsResolving or not myHRP or not kHRP then
                            if tracker then tracker:Disconnect() end
                            return
                        end
                        local currentDist = (myHRP.Position - kHRP.Position).Magnitude
                        if currentDist <= aggressiveRadius then
                            DoParry()
                            if tracker then tracker:Disconnect() end
                        end
                    end)
                end
            else
                if startDistance > range then return end
                local myPosFlat = Vector3.new(myHRP.Position.X, 0, myHRP.Position.Z)
                local kPosFlat = Vector3.new(kHRP.Position.X, 0, kHRP.Position.Z)
                local flatDelta = myPosFlat - kPosFlat
                if flatDelta.Magnitude > 0 then
                    local flatDirection = flatDelta.Unit
                    local kLookFlat = Vector3.new(kHRP.CFrame.LookVector.X, 0, kHRP.CFrame.LookVector.Z).Unit
                    local isFacing = kLookFlat:Dot(flatDirection)
                    if isFacing < 0.6 then return end
                end
                DoParry()
            end
        end)
    end

    local function VD_AngularDelta(from, to)
        local d = to - from
        if d > 180 then d = d - 360 end
        if d < -180 then d = d + 360 end
        return d
    end

    local function VD_CrossedZone(prevLr, lr, startPos, endPos)
        local function inZone(r)
            if startPos > endPos then
                return r >= startPos or r <= endPos
            end
            return r >= startPos and r <= endPos
        end
        if inZone(lr) then return true end
        if prevLr == nil then return false end
        local delta = VD_AngularDelta(prevLr, lr)
        local steps = math.abs(math.floor(delta))
        if steps < 2 then return false end
        local stepSize = delta / steps
        for i = 1, steps do
            if inZone((prevLr + stepSize * i) % 360) then return true end
        end
        return false
    end

    local function VD_NormalSkillcheckUpdate()
        local line, goal = VD_GetSkillCheck()
        if not (line and goal) then
            AutoSkill.LastGoalRotation = nil
            AutoSkill.HasClickedThisGoal = false
            AutoSkill.LastLineRotation = nil
            AutoSkill.LastTick = nil
            AutoSkill.WasActive = false
            return
        end

        local lr = line.Rotation % 360
        local gr = goal.Rotation % 360
        local now = os.clock()
        if not AutoSkill.WasActive then
            AutoSkill.WasActive = true
            AutoSkill.HasClickedThisGoal = false
            AutoSkill.LastGoalRotation = gr
            AutoSkill.LastLineRotation = lr
            AutoSkill.LastTick = now
            return
        end
        if AutoSkill.LastGoalRotation and math.abs(VD_AngularDelta(AutoSkill.LastGoalRotation, gr)) > 5 then
            AutoSkill.HasClickedThisGoal = false
            AutoSkill.LastLineRotation = nil
            AutoSkill.LastTick = nil
        end
        AutoSkill.LastGoalRotation = gr
        if AutoSkill.HasClickedThisGoal then
            AutoSkill.LastLineRotation = lr
            AutoSkill.LastTick = now
            return
        end
        if AutoSkill.LastLineRotation and AutoSkill.LastTick then
            local dt = now - AutoSkill.LastTick
            if dt > 0 then
                local lineSpeed = VD_AngularDelta(AutoSkill.LastLineRotation, lr) / dt
                local predicted = (lr + lineSpeed * dt * 0) % 360
                if VD_CrossedZone(AutoSkill.LastLineRotation, predicted, (gr + 104) % 360, (gr + 109) % 360) then
                    AutoSkill.HasClickedThisGoal = true
                    task.spawn(function()
                        task.wait(0.03)
                        VD_PressSkill()
                    end)
                end
            end
        end
        AutoSkill.LastLineRotation = lr
        AutoSkill.LastTick = now
    end

        if AutoSkill.PerfectLastGoalRotation and math.abs(VD_AngularDelta(AutoSkill.PerfectLastGoalRotation, gr)) > 5 then
            AutoSkill.PerfectHasClickedThisGoal = false
            AutoSkill.PerfectLastLineRotation = nil
            AutoSkill.PerfectLastTick = nil
        end
        AutoSkill.PerfectLastGoalRotation = gr
        if AutoSkill.PerfectHasClickedThisGoal then
            AutoSkill.PerfectLastLineRotation = lr
            AutoSkill.PerfectLastTick = now
            return
        end
        if AutoSkill.PerfectLastLineRotation and AutoSkill.PerfectLastTick then
            local dt = now - AutoSkill.PerfectLastTick
            if dt > 0 then
                local lineSpeed = VD_AngularDelta(AutoSkill.PerfectLastLineRotation, lr) / dt
                local predicted = (lr + lineSpeed * dt * 0) % 360
                if VD_CrossedZone(AutoSkill.PerfectLastLineRotation, predicted, (gr + 104) % 360, (gr + 108) % 360) then
                    AutoSkill.PerfectHasClickedThisGoal = true
                    VD_PressSkill()
                end
            end
        end
        AutoSkill.PerfectLastLineRotation = lr
        AutoSkill.PerfectLastTick = now
    end

    local function VD_InstantSkillcheckUpdate()
        local line, goal = VD_GetSkillCheck()
        if not (line and goal) then
            AutoSkill.InstantHasClicked = false
            AutoSkill.InstantLastGoalRotation = 0
            AutoSkill.InstantLastGoalInstance = nil
            AutoSkill.InstantCurrentGoalID = 0
            if AutoSkill.InstantRotationConnection then
                AutoSkill.InstantRotationConnection:Disconnect()
                AutoSkill.InstantRotationConnection = nil
            end
            return
        end

        local gr = goal.Rotation % 360
        local perfectRot = (gr + 106) % 360
        if not AutoSkill.InstantForcingRotation then
            AutoSkill.InstantForcingRotation = true
            pcall(function() line.Rotation = perfectRot end)
            AutoSkill.InstantForcingRotation = false
        end

        local diff = math.abs(gr - AutoSkill.InstantLastGoalRotation)
        if diff > 180 then diff = 360 - diff end
        local isNewGoal = diff > 0.5 or AutoSkill.InstantLastGoalInstance ~= goal
        if isNewGoal then
            AutoSkill.InstantHasClicked = false
            AutoSkill.InstantCurrentGoalID = AutoSkill.InstantCurrentGoalID + 1
            local assignedID = AutoSkill.InstantCurrentGoalID
            if AutoSkill.InstantRotationConnection then AutoSkill.InstantRotationConnection:Disconnect() end
            AutoSkill.InstantRotationConnection = line:GetPropertyChangedSignal("Rotation"):Connect(function()
                if AutoSkill.InstantForcingRotation then return end
                AutoSkill.InstantForcingRotation = true
                pcall(function()
                    local _, cGoal = VD_GetSkillCheck()
                    if cGoal then line.Rotation = (cGoal.Rotation % 360 + 106) % 360 end
                end)
                AutoSkill.InstantForcingRotation = false
            end)
            if not AutoSkill.InstantHasClicked then
                AutoSkill.InstantHasClicked = true
                task.spawn(function()
                    task.wait(0.05)
                    if AutoSkill.InstantCurrentGoalID == assignedID then
                        local cl, cg = VD_GetSkillCheck()
                        if cl and cg and tick() - AutoSkill.InstantLastTriggerTick > 0.03 then
                            AutoSkill.InstantLastTriggerTick = tick()
                            VD_PressSkill()
                        end
                    end
                end)
            end
        end
        AutoSkill.InstantLastGoalRotation = gr
        AutoSkill.InstantLastGoalInstance = goal
    end

    local ORIGINAL_WALKSPEED = 19
    local originalWalkSpeed = ORIGINAL_WALKSPEED

    local function GetCrouchButton()
        local gui = LocalPlayer:FindFirstChild("PlayerGui")
        if not gui then return nil end
        local survivorMob = gui:FindFirstChild("Survivor-mob")
        if not survivorMob then return nil end

        return controls:FindFirstChild("crouch")

    local activeSlashers = {}
    local animConns = {}
    local currentCrouchState = false

    local function SetCrouchSync(state)
        if currentCrouchState == state then return end
        currentCrouchState = state

        local char = LocalPlayer.Character
        if char then
            char:SetAttribute("Crouching", state)
            char:SetAttribute("Crouchingserver", state)

            local humanoid = char:FindFirstChildOfClass("Humanoid")
            if humanoid then
                if originalWalkSpeed == ORIGINAL_WALKSPEED and humanoid.WalkSpeed > 0 then
                    originalWalkSpeed = humanoid.WalkSpeed
                end
                if state then
                    humanoid.WalkSpeed = 2
                else
                    humanoid.WalkSpeed = originalWalkSpeed or ORIGINAL_WALKSPEED
                end
            end
        end

        pcall(function()
            local remotes = ReplicatedStorage:FindFirstChild("Remotes")
            if remotes then
                local mech = remotes:FindFirstChild("Mechanics")
                if mech then
                    local ca = mech:FindFirstChild("ChangeAttribute")
                    if ca then
                        ca:FireServer("Crouching", state)
                        ca:FireServer("Crouchingserver", state)
                    end
                end
            end
        end)

        local isPC = not UserInputService.TouchEnabled and UserInputService.KeyboardEnabled
        if isPC then
            VirtualInputManager:SendKeyEvent(state, Enum.KeyCode.C, false, game)
            VirtualInputManager:SendKeyEvent(state, Enum.KeyCode.LeftControl, false, game)
        else
            local btn = GetCrouchButton()
            if btn and type(firesignal) == "function" then
                if state then
                    firesignal(btn.MouseButton1Down)
                    if btn.MouseButton1Click then firesignal(btn.MouseButton1Click) end
                else
                    firesignal(btn.MouseButton1Up)
                    if btn.MouseButton1Click then firesignal(btn.MouseButton1Click) end
                end
            end
        end
    end

    local function HookSlasherAnimation(character)
        if not character then return end
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if not humanoid then return end
        if animConns[humanoid] then return end

        animConns[humanoid] = humanoid.AnimationPlayed:Connect(function(animTrack)
            if not getgenv().VD.SURV_AutoCrouch then return end
            local anim = animTrack.Animation
            if not anim then return end
            local animId = tostring(anim.AnimationId):match("%d+") or ""
            if animId == "80411309607666" then
                activeSlashers[character] = true
                animTrack.Stopped:Connect(function()
                    task.wait(0.2)
                    activeSlashers[character] = nil
                end)
            end
        end)

        humanoid.Died:Connect(function()
            activeSlashers[character] = nil
        end)
    end

    local function UpdateAutoCrouch()
        local vd = getgenv().VD
        if not vd.SURV_AutoCrouch then
            if currentCrouchState then
                SetCrouchSync(false)
            end
            return
        end

        local char = LocalPlayer.Character
        if not char then
            if currentCrouchState then
                SetCrouchSync(false)
            end
            return
        end

        local playerPos = char:FindFirstChild("HumanoidRootPart")
        if not playerPos then
            if currentCrouchState then
                SetCrouchSync(false)
            end
            return
        end

        local radius = vd.SURV_AutoCrouchRadius or 18
        local shouldCrouch = false

        for slasherChar, _ in pairs(activeSlashers) do
            if slasherChar and slasherChar.Parent then
                local root = slasherChar:FindFirstChild("HumanoidRootPart")
                if root then
                    local dist = (root.Position - playerPos.Position).Magnitude
                    if dist <= radius then
                        shouldCrouch = true
                        break
                    end
                end
            else
                activeSlashers[slasherChar] = nil
            end
        end

        if currentCrouchState ~= shouldCrouch then
            SetCrouchSync(shouldCrouch)
        end
    end

    task.spawn(function()
        while task.wait(0.15) do
            pcall(UpdateAutoCrouch)
        end
    end)

    -- AUTO UNHOOK
    local function nearestTagged(tag, maxDistance)
        local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if not root then return nil, math.huge end
        local best, bestDistance = nil, maxDistance or math.huge
        local own = LocalPlayer.Character
        for _, inst in ipairs(CollectionService:GetTagged(tag)) do
            if inst:IsA("BasePart") and inst:IsDescendantOf(Workspace) and not (own and inst:IsDescendantOf(own)) then
                local blocked = inst.Parent and CollectionService:HasTag(inst.Parent, "Blocked")
                if not blocked then
                    local d = (inst.Position - root.Position).Magnitude
                    if d < bestDistance then
                        best, bestDistance = inst, d
                    end
                end
            end
        end
        return best, bestDistance
    end

    local function faceCFrame(fromPosition, targetPosition)
        return CFrame.lookAt(fromPosition, Vector3.new(targetPosition.X, fromPosition.Y, targetPosition.Z))
    end

    local function safeTeleport(cframe)
        local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if not root or not cframe then return false end
        pcall(function() root.CFrame = cframe end)
        return true
    end

    local function settle(seconds)
        task.wait(seconds or 0.35)
        return true
    end

    local function autoUnhookTick()
        if not getgenv().VD.SURV_AutoUnhook then return end
        if GetRole(LocalPlayer) ~= "Survivor" or not isAlive() then return end
        local c = LocalPlayer.Character
        if not c then return end
        if c:GetAttribute("IsHooked") or c:GetAttribute("IsCarried") or c:GetAttribute("Knocked") then return end
        local ci = c:FindFirstChild("CheckInterractable")
        if ci and ci:GetAttribute("isUnhooking") then return end

        local point, dist = nearestTagged("UnhookPoint", 150)
        if not point then return end

        local root = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if not root then return end

        if dist > 3.5 then
            safeTeleport(faceCFrame(point.Position + Vector3.new(0, 0, 2), point.Position))
            if not settle(0.35) then return end
        end

        if ci then ci:SetAttribute("isUnhooking", true) end
        local unhookRemote = ReplicatedStorage:FindFirstChild("Remotes") and ReplicatedStorage.Remotes:FindFirstChild("Carry") and ReplicatedStorage.Remotes.Carry:FindFirstChild("UnHookEvent")
        if unhookRemote then
            pcall(function() unhookRemote:FireServer(point) end)
        end
        task.delay(2.2, function()
            local c2 = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("CheckInterractable")
            if c2 then c2:SetAttribute("isUnhooking", false) end
        end)
    end

    task.spawn(function()
        while task.wait(0.3) do
            pcall(autoUnhookTick)
        end
    end)

                if gate.Name == "Gate" then
                    table.insert(gates, gate)
                end
            end
        end
        return gates

    local function setBypassGate(state)
        local gates = gatherGates()
        for _, gate in pairs(gates) do
            local leftGate = gate:FindFirstChild("LeftGate")
            local rightGate = gate:FindFirstChild("RightGate")
            local leftEnd = gate:FindFirstChild("LeftGate-end")
            local rightEnd = gate:FindFirstChild("RightGate-end")
            local box = gate:FindFirstChild("Box")
            if state then
                if leftGate then
                    leftGate.Transparency = 1
                    leftGate.CanCollide = false
                end
                if rightGate then
                    rightGate.Transparency = 1
                    rightGate.CanCollide = false
                end
                if leftEnd then
                    leftEnd.Transparency = 0
                    leftEnd.CanCollide = true
                end
                if rightEnd then
                    rightEnd.Transparency = 0
                    rightEnd.CanCollide = true
                end
                if box then
                    box.CanCollide = false
                end
            else
                if leftGate then
                    leftGate.Transparency = 0
                    leftGate.CanCollide = true
                end
                if rightGate then
                    rightGate.Transparency = 0
                    rightGate.CanCollide = true
                end
                if leftEnd then
                    leftEnd.Transparency = 1
                    leftEnd.CanCollide = true
                end
                if rightEnd then
                    rightEnd.Transparency = 1
                    rightEnd.CanCollide = true
                end
                if box then
                    box.CanCollide = true
                end
            end
        end
    end

        FlashlightPart = nil,

    local function IsValidTarget(character)
        if not character then return false end
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if not humanoid or humanoid.Health <= 50 then return false end
        local hooked = character:GetAttribute("IsHooked") or character:GetAttribute("isHooked")

    local function GetFlashlightActivateRemote()
        local remotes = ReplicatedStorage:FindFirstChild("Remotes")
        local items = remotes and remotes:FindFirstChild("Items")
        local flashlight = items and items:FindFirstChild("Flashlight")

                return activate
            end
        end

    end
    local flashActivateRemote = GetFlashlightActivateRemote()

    local function GetFlashlightTargetPart(char)
        if not char then return nil end
        local pref = VD.FLASH_TargetPart or "Head"
        local part = char:FindFirstChild(pref)
        if part and part:IsA("BasePart") then return part end
        return char:FindFirstChild("Head")
            or char:FindFirstChild("UpperTorso")
            or char:FindFirstChild("Torso")
            or char:FindFirstChild("HumanoidRootPart")
    end

    local function GetFlashlightTarget()
        local localChar = LocalPlayer.Character
        local localRoot = localChar and localChar:FindFirstChild("HumanoidRootPart")
        if not localRoot then return nil end

        local maxRange = tonumber(VD.FLASH_Range) or 120
        local bestPart, bestDist = nil, math.huge

        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character and IsValidTarget(player.Character) then
                local isKiller = player.Team and player.Team.Name == "Killer"
                if isKiller then
                    local part = GetFlashlightTargetPart(player.Character)
                    if part then
                        local dist = (localRoot.Position - part.Position).Magnitude
                        if dist <= maxRange and dist < bestDist then
                            bestDist = dist
                            bestPart = part
                        end
                    end
                end
            end
        end
        return bestPart
    end

    local function ClearFlashlightLaser()
        if FlashlightAimState.LaserBeam then
            pcall(function() FlashlightAimState.LaserBeam:Destroy() end)
            FlashlightAimState.LaserBeam = nil
        end
    end

    local function GetFlashlightOrigin(cam)
        local source = FlashlightAimState.FlashlightPart
        if source and source:IsA("BasePart") then
            return source.Position
        elseif source then
            local part = source:FindFirstChildWhichIsA("BasePart", true)
            if part then return part.Position end
        end
        local char = LocalPlayer.Character
        local hand = char and (char:FindFirstChild("RightHand")
            or char:FindFirstChild("Right Arm")
            or char:FindFirstChild("HumanoidRootPart"))
        if hand and hand:IsA("BasePart") then
            return hand.Position
        end
        return cam and cam.CFrame.Position or nil
    end

    local function UpdateFlashlightLaser(originPos, targetPos)
        if not VD.FLASH_Laser then
            if FlashlightAimState.LaserBeam then
                FlashlightAimState.LaserBeam.Transparency = 1
            end
            return
        end

        if not FlashlightAimState.LaserBeam then
            local laser = Instance.new("Part")
            laser.Name = "FlashlightSilentAimLaser"
            laser.Anchored = true
            laser.CanCollide = false
            laser.CanTouch = false
            laser.CastShadow = false
            laser.Material = Enum.Material.Neon
            laser.Color = Color3.fromRGB(80, 220, 255)
            laser.Transparency = 0
            laser.Parent = Workspace
            FlashlightAimState.LaserBeam = laser
        end

        local dist = (targetPos - originPos).Magnitude
        if dist < 0.1 then return end

        local laser = FlashlightAimState.LaserBeam
        laser.Size = Vector3.new(0.05, 0.05, dist)
        laser.CFrame = CFrame.new((originPos + targetPos) / 2, targetPos)
        laser.Transparency = 0
    end

    local function FlashlightAimStep()
        local hasFlash = HasFlashlight()
        local isSurv = IsSurvivorTeam()

        if not (hasFlash and isSurv and VD.FLASH_SilentAim and FlashlightAimState.Active) then
            if FlashlightAimState.LaserBeam then
                FlashlightAimState.LaserBeam.Transparency = 1
            end
            return
        end

        local cam = Workspace.CurrentCamera
        local targetPart = GetFlashlightTarget()
        if not (cam and targetPart) then
            if FlashlightAimState.LaserBeam then
                FlashlightAimState.LaserBeam.Transparency = 1
            end
            return
        end

        local targetPos = targetPart.Position
        local smooth = math.clamp(tonumber(VD.FLASH_Smooth) or 0.35, 0.05, 1)
        local originPos = GetFlashlightOrigin(cam)

        if VD.FLASH_Laser and originPos then
            UpdateFlashlightLaser(originPos, targetPos)
        elseif FlashlightAimState.LaserBeam then
            FlashlightAimState.LaserBeam.Transparency = 1
        end

        pcall(function()
            cam.CFrame = cam.CFrame:Lerp(CFrame.new(cam.CFrame.Position, targetPos), smooth)
        end)

        pcall(function()
            local char = LocalPlayer.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            if hrp then
                hrp.CFrame = CFrame.new(hrp.Position, Vector3.new(targetPos.X, hrp.Position.Y, targetPos.Z))
            end
        end)
    end

    local function StartFlashlightSilentAim()
        if FlashlightAimState.Connection then return end
        FlashlightAimState.Connection = RunService.RenderStepped:Connect(FlashlightAimStep)
    end

    local function StopFlashlightSilentAim()
        FlashlightAimState.Active = false
        FlashlightAimState.FlashlightPart = nil
        ClearFlashlightLaser()
        if FlashlightAimState.Connection then
            pcall(function() FlashlightAimState.Connection:Disconnect() end)
            FlashlightAimState.Connection = nil
        end
    end

    local function SetFlashlightSilentAim(enabled)

        if VD.FLASH_SilentAim then
            StartFlashlightSilentAim()
        else
            StopFlashlightSilentAim()
        end

            if FlaskAimState.LaserPart then
                FlaskAimState.LaserPart.Transparency = 1
            end
            return
        end

        local targetPos = nil
        local minDist = math.huge
        local originPos = nil

        local hand = char:FindFirstChild("LeftHand") or char:FindFirstChild("Left Arm")
        originPos = hand and hand.Position or hrp.Position

        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local targetChar = player.Character
                if IsValidTarget(targetChar) and not (targetChar:GetAttribute("IsKiller") == true) then
                    local tPos = GetFlashlightTargetPart(targetChar)
                    if tPos then
                        local dist = (hrp.Position - tPos.Position).Magnitude
                        if dist < minDist then
                            minDist = dist
                            targetPos = tPos.Position
                        end
                    end
                end
            end
        end

        if not (originPos and targetPos) then
            if FlaskAimState.LaserPart then
                FlaskAimState.LaserPart.Transparency = 1
            end
            return
        end

        if not FlaskAimState.LaserPart then
            local laser = Instance.new("Part")
            laser.Name = "FlaskSilentAimLaser"
            laser.Anchored = true
            laser.CanCollide = false
            laser.CanTouch = false
            laser.CastShadow = false
            laser.Material = Enum.Material.Neon
            laser.Color = Color3.fromRGB(0, 100, 255)
            laser.Transparency = 0
            laser.Parent = Workspace
            FlaskAimState.LaserPart = laser
        end

        local dist = (targetPos - originPos).Magnitude
        if dist > 0.1 then
            local laser = FlaskAimState.LaserPart
            laser.Size = Vector3.new(0.05, 0.05, dist)
            laser.CFrame = CFrame.new((originPos + targetPos) / 2, targetPos)
            laser.Transparency = 0
        else
            FlaskAimState.LaserPart.Transparency = 1
        end
    end

    local function SetupCombinedHooks()
        local mt = getrawmetatable(game)
        if not mt then return end
        local oldNamecall = mt.__namecall

        local newNamecall = function(self, ...)
            local method = getnamecallmethod()
            local args = {...}

            if method == "FireServer" and self == antiBlindRemote and VD.KILLER_AntiBlind and GetRole(LocalPlayer) == "Killer" then
                return nil
            end

            if method == "FireServer" and self.Name == "ThrowFlask" and VD.KILLER_SilentAimFlask then
                local lp = LocalPlayer
                local myPos = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart") and lp.Character.HumanoidRootPart.Position
                local closest, minDist = nil, math.huge
                if myPos then
                    for _, player in ipairs(Players:GetPlayers()) do
                        if player ~= lp and player.Character then
                            local targetChar = player.Character
                            if IsValidTarget(targetChar) and not (targetChar:GetAttribute("IsKiller") == true) then
                                local tPos = GetFlashlightTargetPart(targetChar)
                                if tPos then
                                    local dist = (myPos - tPos.Position).Magnitude
                                    if dist < minDist then
                                        minDist = dist
                                        closest = tPos.Position
                                    end
                                end
                            end
                        end
                    end
                end
                if closest then
                    if args[2] and typeof(args[2]) == "Vector3" then
                        args[1] = (closest - args[2]).Unit
                    end
                end
                return oldNamecall(self, unpack(args))
            end

            if flashActivateRemote and self == flashActivateRemote then
                local params = {...}
                pcall(SetFlashlightAimActive, params[2] == true, params[1])
            end

            if method == "FireServer" and not checkcaller() then
                if VD.KILLER_InfFrenzy and self.Name == "Deactivatefromclient" then
                    local ok, parent = pcall(function() return self.Parent end)
                    if ok and parent and parent.Name == "Killer" then
                        return nil
                    end
                end

                if VD.KILLER_InfLakeMist and self.Name == "LakeMist" then
                    if args[1] == false then
                        return nil
                    end
                end
                if VD.KILLER_InfPursuit and self.Name == "Pursuit" then
                    if args[1] == false then
                        return nil
                    end
                end
            end

            return oldNamecall(self, ...)
        end

        setreadonly(mt, false)
        mt.__namecall = newNamecall
        setreadonly(mt, true)
    end

    pcall(SetupCombinedHooks)

        local function ESP_GetPlayerRole(player)
            return ESP_State.playerRoles[player] or "survivor"

        local function ESP_UpdatePlayerRole(player)
            local ok, teamName = pcall(function()
                return player.Team and player.Team.Name:lower() or ""
            end)
            if ok then
                if teamName:find("killer") then
                    ESP_State.playerRoles[player] = "killer"
                elseif teamName:find("spectator") then
                    ESP_State.playerRoles[player] = "spectator"
                else
                    ESP_State.playerRoles[player] = "survivor"
                end
            else
                ESP_State.playerRoles[player] = "survivor"
            end
        end

        local function ESP_ApplyHighlight(instance, color)
            if not instance or not instance.Parent then return end
            local h = instance:FindFirstChild("__GlutoHL__")
            if not h then
                h = Instance.new("Highlight")
                h.Name = "__GlutoHL__"
                h.Adornee = instance
                h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                h.FillTransparency = ESP_Settings.fillTransparency
                h.OutlineTransparency = 0.03
                h.LineThickness = 2
                h.FillColor = color or Color3.fromRGB(180, 180, 180)
                h.OutlineColor = Color3.fromRGB(255, 255, 255)
                h.Parent = instance
            else
                h.FillColor = color or Color3.fromRGB(180, 180, 180)
                h.OutlineColor = Color3.fromRGB(255, 255, 255)
                if not h.Enabled then h.Enabled = true end
            end
        end

        local function ESP_RemoveHighlight(instance)
            if not instance then return end
            local h = instance:FindFirstChild("__GlutoHL__")
            if h then h:Destroy() end
        end

        local function ESP_GetRichNameText(player)
            local displayName = player.DisplayName or player.Name

            local roleText = role == "killer" and "Killer" or (role == "spectator" and "Spectator" or "Survivors")
            local roleColor = role == "killer" and "128,0,0" or (role == "spectator" and "180,180,180" or "0,128,0")
            return string.format(
                "<font color='rgb(255,255,255)'>%s</font> [<font color='rgb(%s)'>%s</font>]",
                displayName,
                roleColor,
                roleText
            )
        end

            local billboard = Instance.new("BillboardGui")

            billboard.AlwaysOnTop = true
            billboard.Size = size or UDim2.new(0, 150, 0, 40)
            billboard.LightInfluence = 0
            billboard.ZIndexBehavior = Enum.ZIndexBehavior.Sibling

            local label = Instance.new("TextLabel")

            label.Text = text

            stroke.Thickness = 1.2
            stroke.Color = Color3.new(0, 0, 0)
            stroke.Transparency = 0.2
            stroke.Parent = label

            label.Parent = billboard
            return billboard
        end

            for player, entry in pairs(ESP_State.espObjects) do
                if entry.billboard and entry.billboard.Parent then
                    local role = ESP_GetPlayerRole(player)

                    entry.nameLabel.Text = fullText
                    entry.nameLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
                    local showName = (role == "killer" and getgenv().ESP_Killer_Name) or
                                     (role == "spectator" and getgenv().ESP_Spectator_Name) or
                                     (role == "survivor" and getgenv().ESP_Survivor_Name)
                    entry.nameLabel.Visible = showName
                    entry.billboard.MaxDistance = ESP_Settings.maxDistance
                end

            for player, highlight in pairs(ESP_State.outlineObjects) do

                    highlight.FillTransparency = ESP_Settings.fillTransparency
                end
            end
        end

        local function ESP_ApplyBillboardToCharacter(player, character)
            local entry = ESP_State.espObjects[player]
            if not entry then return end
            if not character or not character.Parent or player.Character ~= character then return end

            local rootPart = character:FindFirstChild("HumanoidRootPart")
                or character:FindFirstChild("Torso")
                or character:FindFirstChild("UpperTorso")
                or character:FindFirstChild("Head")

            if not rootPart then
                task.spawn(function()
                    local ok, part = pcall(function()
                        return character:WaitForChild("HumanoidRootPart", 3)
                    end)
                    if ok and part and part.Parent and character.Parent and player.Character == character then
                        entry.billboard.Adornee = part
                        entry.billboard.Enabled = true
                    end
                end)
                return
            end

            entry.billboard.Adornee = rootPart
            entry.billboard.Enabled = true

            local role = ESP_GetPlayerRole(player)
            local fullText = ESP_GetRichNameText(player)
            entry.nameLabel.Text = fullText
            entry.nameLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
            local showName = (role == "killer" and getgenv().ESP_Killer_Name) or
                             (role == "spectator" and getgenv().ESP_Spectator_Name) or
                             (role == "survivor" and getgenv().ESP_Survivor_Name)
            entry.nameLabel.Visible = showName
        end

        local function ESP_ValidatePlayer(player)
            if player == LocalPlayer or not player.Parent then return end

            local billboardEntry = ESP_State.espObjects[player]
            local highlight = ESP_State.outlineObjects[player]
            if not billboardEntry and not highlight then return end

            local character = player.Character
            if not character or not character.Parent then
                if billboardEntry then billboardEntry.billboard.Adornee = nil end
                if highlight then
                    highlight.Adornee = nil
                    highlight.Enabled = false
                end
                return
            end

            local rootPart = character:FindFirstChild("HumanoidRootPart")
                or character:FindFirstChild("Torso")
                or character:FindFirstChild("UpperTorso")
                or character:FindFirstChild("Head")
            if not rootPart then return end

            if billboardEntry then
                if billboardEntry.billboard.Adornee ~= rootPart then
                    billboardEntry.billboard.Adornee = rootPart
                end
                if not billboardEntry.billboard.Enabled then
                    billboardEntry.billboard.Enabled = true
                end
            end

            if highlight then
                if highlight.Adornee ~= character then
                    ESP_ApplyOutlineToCharacter(player, character)
                elseif not highlight.Enabled then
                    local role = ESP_GetPlayerRole(player)
                    highlight.Enabled = (role == "killer" and getgenv().ESP_Killer_Highlight) or
                                        (role == "spectator" and getgenv().ESP_Spectator_Highlight) or
                                        (role == "survivor" and getgenv().ESP_Survivor_Highlight)
                end
            end
        end

        -- OBJECT ESP
        local function ESP_UpdateGeneratorProgress(generator)
            if not generator or not generator.Parent then return true end
            if ESP_State.completedGenerators[generator] then return true end

            local progress = generator:GetAttribute("RepairProgress") or generator:GetAttribute("Progress") or 0
            local isDone = (progress >= 100)
                or (generator:GetAttribute("Completed") == true)
                or (generator:GetAttribute("IsCompleted") == true)
                or (generator:GetAttribute("Done") == true)

            local progressGui = generator:FindFirstChild("__GlutoGenProgress__")

            if isDone then
                if progressGui then progressGui:Destroy() end
                ESP_RemoveHighlight(generator)
                generator:SetAttribute("__GlutoGenLastPct__", nil)
                ESP_State.completedGenerators[generator] = true
                ESP_State.genIndices[generator] = nil
                return true
            end

            if getgenv().ESP_Generator then
                ESP_ApplyHighlight(generator, Color3.fromRGB(180, 180, 180))
            else
                ESP_RemoveHighlight(generator)
            end

            if not getgenv().ESP_Generator then
                if progressGui then
                    progressGui:Destroy()
                    generator:SetAttribute("__GlutoGenLastPct__", nil)
                end
                return false
            end

            local roundedPct = math.floor(progress + 0.5)
            local lastPct = generator:GetAttribute("__GlutoGenLastPct__")
            if lastPct == roundedPct and progressGui then
                return false
            end
            generator:SetAttribute("__GlutoGenLastPct__", roundedPct)

            local anchorPart = generator:FindFirstChild("GeneratorBody", true)
                or generator:FindFirstChild("defaultMaterial", true)
                or (generator:IsA("Model") and generator.PrimaryPart)
                or generator:FindFirstChildWhichIsA("BasePart", true)
            if not anchorPart then return false end

            local percentStr = roundedPct .. "%"
            if not progressGui then
                progressGui = Instance.new("BillboardGui")
                progressGui.Name = "__GlutoGenProgress__"
                progressGui.Adornee = anchorPart
                progressGui.AlwaysOnTop = true
                progressGui.LightInfluence = 0
                progressGui.ResetOnSpawn = false
                progressGui.MaxDistance = 260
                progressGui.Size = UDim2.new(0, 100, 0, 14)
                progressGui.StudsOffset = Vector3.new(0, (anchorPart.Size.Y / 2) + 3.5, 0)
                progressGui.Parent = generator

                local label = Instance.new("TextLabel")
                label.Name = "Label"
                label.BackgroundTransparency = 1
                label.Size = UDim2.new(1, 0, 1, 0)
                label.Font = Enum.Font.GothamBlack
                label.TextSize = 11
                label.RichText = false
                label.Text = percentStr
                label.TextColor3 = Color3.fromRGB(255, 255, 255)
                label.TextXAlignment = Enum.TextXAlignment.Center
                label.Parent = progressGui

                local stroke = Instance.new("UIStroke")
                stroke.Thickness = 1
                stroke.Transparency = 0.2
                stroke.Color = Color3.new(0, 0, 0)
                stroke.Parent = label
            else
                if progressGui.Adornee ~= anchorPart then
                    progressGui.Adornee = anchorPart
                end
                progressGui.StudsOffset = Vector3.new(0, (anchorPart.Size.Y / 2) + 3.5, 0)
                local label = progressGui:FindFirstChild("Label")
                if label then label.Text = percentStr end
            end

            return false
        end

        local function ESP_IsSCP(v)
            if not (v and v:IsA("Model")) then return false end
            local n = v.Name:lower()
            return n == "scp"
                or n:match("^scp%d*$")
                or n:match("^scp[%-%_]?%d+$")
                or n:find("zombie")
                or n:find("monster")
                or n:find("infected")
                or n:find("mutant")
        end

        local function ESP_RemoveSCP(v)
            local h = ESP_State.scpCache[v]
            if h then pcall(function() h:Destroy() end) end
            ESP_State.scpCache[v] = nil
        end

        local function ESP_CreateSCP(v)
            if not getgenv().ESP_Enable or not getgenv().ESP_SCP or ESP_State.scpCache[v] or not (v and v.Parent) or not ESP_IsSCP(v) then
                return
            end
            local root = v:FindFirstChild("HumanoidRootPart", true) or v.PrimaryPart or v:FindFirstChildWhichIsA("BasePart", true)
            if not root then return end

            local h = Instance.new("Highlight")
            h.Name = "SCPESP"
            h.Adornee = v
            h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            h.FillColor = Color3.fromRGB(180, 180, 180)
            h.OutlineColor = Color3.fromRGB(255, 255, 255)
            h.FillTransparency = 0.78
            h.OutlineTransparency = 0.03
            h.Parent = ESPFolder
            ESP_State.scpCache[v] = h

            v.AncestryChanged:Connect(function(_, p)
                if not p then ESP_RemoveSCP(v) end
            end)
            local hum = v:FindFirstChildOfClass("Humanoid")
            if hum then
                hum.Died:Connect(function()
                    ESP_RemoveSCP(v)
                end)
            end
        end

        local function ESP_ScanSCP()
            for _, v in ipairs(workspace:GetChildren()) do
                if ESP_IsSCP(v) then ESP_CreateSCP(v) end
            end
        end

        local function ESP_GetWindowBottomPart(instance)
            if typeof(instance) ~= "Instance" then return nil end
            local lowerName = string.lower(instance.Name)
            if lowerName == "window" and instance:IsA("Model") then
                local bottom = instance:FindFirstChild("Bottom", true)
                if bottom and bottom:IsA("BasePart") then
                    return bottom
                end
                return instance.PrimaryPart
            end
            if lowerName == "bottom" and instance:IsA("BasePart") then
                if instance.Parent and string.lower(instance.Parent.Name) == "window" then
                    return instance
                end
            end
            return nil
        end

            end
        end

        local function ESP_OnMapDescendantAdded(instance)
            local name = instance.Name

            if name == "Generator" then
                table.insert(ESP_State.cachedMapObjects.Generators, instance)
                if not ESP_State.genIndices[instance] then
                    if next(ESP_State.genIndices) == nil then
                        ESP_State.nextGenIndex = 1
                    end
                    ESP_State.genIndices[instance] = ESP_State.nextGenIndex
                    ESP_State.nextGenIndex = ESP_State.nextGenIndex + 1
                end
                if getgenv().ESP_Generator then
                    ESP_UpdateGeneratorProgress(instance)
                end

            elseif name == "Hook" then
                table.insert(ESP_State.cachedMapObjects.Hooks, instance)
                local meshParts = {}
                local modelChild = instance:FindFirstChild("Model")
                if modelChild then
                    for _, child in ipairs(modelChild:GetDescendants()) do
                        if child:IsA("MeshPart") then
                            table.insert(meshParts, child)
                        end
                    end
                end
                ESP_State.cachedHookMeshParts[instance] = meshParts
                if getgenv().ESP_Hook then
                    if #meshParts > 0 then
                        for _, part in ipairs(meshParts) do
                            ESP_ApplyHighlight(part, Color3.fromRGB(180, 180, 180))
                        end
                    else
                        ESP_ApplyHighlight(instance, Color3.fromRGB(180, 180, 180))
                    end
                end

            elseif name == "Gate" then
                table.insert(ESP_State.cachedMapObjects.Gates, instance)
                if getgenv().ESP_Gate then
                    ESP_ApplyHighlight(instance, Color3.fromRGB(180, 180, 180))
                end

            elseif name == "Pallet" or name == "Palletwrong" then
                table.insert(ESP_State.cachedMapObjects.Pallets, instance)
                local lowerName = name:lower()
                local palletPart = (instance:IsA("Model") and instance.PrimaryPart)
                    or instance:FindFirstChildWhichIsA("BasePart", true)
                    or (instance:IsA("BasePart") and instance)
                local isFake = lowerName:find("fake") or lowerName:find("broken") or lowerName:find("destroyed")
                ESP_State.cachedPalletMeta[instance] = { part = palletPart, isFake = isFake and true or false }
                if getgenv().ESP_Pallet and not isFake then
                    ESP_ApplyHighlight(instance, Color3.fromRGB(180, 180, 180))
                end
            end
        end

        local function ESP_OnMapDescendantRemoving(instance)
            local name = instance.Name

            if name == "Generator" then
                for i, obj in ipairs(ESP_State.cachedMapObjects.Generators) do
                    if obj == instance then
                        table.remove(ESP_State.cachedMapObjects.Generators, i)
                        break
                    end
                end
                ESP_State.genIndices[instance] = nil
                ESP_State.completedGenerators[instance] = nil
                ESP_RemoveHighlight(instance)
                local progressGui = instance:FindFirstChild("__GlutoGenProgress__")
                if progressGui then progressGui:Destroy() end
                if next(ESP_State.genIndices) == nil then
                    ESP_State.nextGenIndex = 1
                end

            elseif name == "Hook" then
                for i, obj in ipairs(ESP_State.cachedMapObjects.Hooks) do
                    if obj == instance then
                        table.remove(ESP_State.cachedMapObjects.Hooks, i)
                        break
                    end
                end
                local meshParts = ESP_State.cachedHookMeshParts[instance]
                if meshParts then
                    for _, part in ipairs(meshParts) do
                        ESP_RemoveHighlight(part)
                    end
                else
                    ESP_RemoveHighlight(instance)
                end
                ESP_State.cachedHookMeshParts[instance] = nil

            elseif name == "Gate" then
                for i, obj in ipairs(ESP_State.cachedMapObjects.Gates) do
                    if obj == instance then
                        table.remove(ESP_State.cachedMapObjects.Gates, i)
                        break
                    end
                end
                ESP_RemoveHighlight(instance)

            elseif name == "Pallet" or name == "Palletwrong" then
                for i, obj in ipairs(ESP_State.cachedMapObjects.Pallets) do
                    if obj == instance then
                        table.remove(ESP_State.cachedMapObjects.Pallets, i)
                        break
                    end
                end
                ESP_RemoveHighlight(instance)
                ESP_State.cachedPalletMeta[instance] = nil
            end
        end

        local function ESP_ScanMapObjects(mapRoot)
            if not mapRoot then return end
            ESP_State.cachedMapObjects = { Generators = {}, Pallets = {}, Hooks = {}, Gates = {} }
            ESP_State.cachedPalletMeta = {}
            ESP_State.cachedHookMeshParts = {}
            ESP_State.completedGenerators = {}
            ESP_State.genIndices = {}
            ESP_State.nextGenIndex = 1
            for _, descendant in ipairs(mapRoot:GetDescendants()) do
                ESP_OnMapDescendantAdded(descendant)
            end
        end

        local function ESP_InitMapObjectESP(mapRoot)
            if not mapRoot then return end
            ESP_ScanMapObjects(mapRoot)
            mapRoot.DescendantAdded:Connect(ESP_OnMapDescendantAdded)
            mapRoot.DescendantRemoving:Connect(ESP_OnMapDescendantRemoving)
            ESP_State.objEspInitialized = true
        end

        local function ESP_RefreshObjectESP()
            if not ESP_State.objEspInitialized then return end

            if getgenv().ESP_Generator then
                local stillActive = {}
                for _, gen in ipairs(ESP_State.cachedMapObjects.Generators) do
                    if gen and gen.Parent then
                        if not ESP_UpdateGeneratorProgress(gen) then
                            table.insert(stillActive, gen)
                        end
                    end
                end
                ESP_State.cachedMapObjects.Generators = stillActive
            else
                for _, gen in ipairs(ESP_State.cachedMapObjects.Generators) do
                    if gen and gen.Parent then
                        ESP_RemoveHighlight(gen)
                        local progressGui = gen:FindFirstChild("__GlutoGenProgress__")
                        if progressGui then progressGui:Destroy() end
                        gen:SetAttribute("__GlutoGenLastPct__", nil)
                    end
                end
            end

            if getgenv().ESP_Pallet then
                for _, pallet in ipairs(ESP_State.cachedMapObjects.Pallets) do
                    if pallet and pallet.Parent then
                        local meta = ESP_State.cachedPalletMeta[pallet]
                        if meta and not meta.isFake then
                            ESP_ApplyHighlight(pallet, Color3.fromRGB(180, 180, 180))
                        end
                    end
                end
            else
                for _, pallet in ipairs(ESP_State.cachedMapObjects.Pallets) do
                    if pallet then ESP_RemoveHighlight(pallet) end
                end
            end

            if getgenv().ESP_Hook then
                for _, hook in ipairs(ESP_State.cachedMapObjects.Hooks) do
                    if hook and hook.Parent then
                        local meshParts = ESP_State.cachedHookMeshParts[hook]
                        if meshParts then
                            for _, part in ipairs(meshParts) do
                                ESP_ApplyHighlight(part, Color3.fromRGB(180, 180, 180))
                            end
                        else
                            ESP_ApplyHighlight(hook, Color3.fromRGB(180, 180, 180))
                        end
                    end
                end
            else
                for _, hook in ipairs(ESP_State.cachedMapObjects.Hooks) do
                    if hook and hook.Parent then
                        local meshParts = ESP_State.cachedHookMeshParts[hook]
                        if meshParts then
                            for _, part in ipairs(meshParts) do
                                ESP_RemoveHighlight(part)
                            end
                        else
                            ESP_RemoveHighlight(hook)
                        end
                    end
                end
            end

            if getgenv().ESP_Gate then
                for _, gate in ipairs(ESP_State.cachedMapObjects.Gates) do
                    if gate and gate.Parent then
                        ESP_ApplyHighlight(gate, Color3.fromRGB(180, 180, 180))
                    end
                end
            else
                for _, gate in ipairs(ESP_State.cachedMapObjects.Gates) do
                    if gate and gate.Parent then ESP_RemoveHighlight(gate) end
                end
            end

            if getgenv().ESP_Window then
                for part, box in pairs(ESP_State.windowEspObjects) do
                    if part and part.Parent and box and box.Parent then
                        box.Color3 = Color3.fromRGB(180, 180, 180)
                        box.Size = part.Size
                    end
                end
                for _, descendant in ipairs(Workspace:GetDescendants()) do
                    local bottomPart = ESP_GetWindowBottomPart(descendant)
                    if bottomPart and getgenv().ESP_Window and not ESP_State.windowEspObjects[bottomPart] then
                        ESP_CreateWindowESP(bottomPart)
                    end
                end
            else
                for part in pairs(ESP_State.windowEspObjects) do
                    ESP_RemoveWindowESP(part)
                end
            end

            if getgenv().ESP_SCP then
                for v, h in pairs(ESP_State.scpCache) do
                    if not (v and v.Parent and h and h.Parent) then
                        ESP_RemoveSCP(v)
                    else
                        if h.Adornee ~= v then h.Adornee = v end
                        h.FillColor = Color3.fromRGB(180, 180, 180)
                        h.OutlineColor = Color3.fromRGB(255, 255, 255)
                        h.FillTransparency = 0.78
                        h.OutlineTransparency = 0.03
                    end
                end
                ESP_ScanSCP()
            else
                for v in pairs(ESP_State.scpCache) do
                    ESP_RemoveSCP(v)
                end
            end
        end

                if VD.AimLockTargetType == "Survivor" and role ~= "survivor" then continue end

                local dist = (targetPos - origin).Magnitude
                if dist < bestDist then
                    bestDist = dist
                    bestPlayer = player
                end
            end
            return bestPlayer
        end

        local function AimLockEnable()
            if aimLockConnection then aimLockConnection:Disconnect() end
            aimLockCachedTarget = nil
            aimLockCachedTargetTime = 0

            aimLockConnection = RunService.RenderStepped:Connect(function(delta)
                if not VD.AimLockEnabled then return end

                if VD.AimLockLockMode == "Hold to Lock" and not aimLockHoldActive then
                    aimLockCachedTarget = nil
                    return
                end

                local char = LocalPlayer.Character
                if not char then return end
                local humanoidRootPart = char:FindFirstChild("HumanoidRootPart")
                if not humanoidRootPart then return end

                aimLockCachedTargetTime = aimLockCachedTargetTime + delta
                if aimLockCachedTargetTime >= 0.15 then
                    aimLockCachedTargetTime = 0
                    aimLockCachedTarget = AimLockGetClosestTarget()
                end

                local targetPlayer = aimLockCachedTarget
                if not targetPlayer then return end

                local targetChar = targetPlayer.Character
                if not targetChar then
                    aimLockCachedTarget = nil
                    return
                end

                local targetHumanoid = targetChar:FindFirstChildOfClass("Humanoid")
                if not targetHumanoid or targetHumanoid.Health <= 0 then
                    aimLockCachedTarget = nil
                    return
                end

                local targetPos = AimLockGetTargetTorsoPosition(targetChar)
                if not targetPos then
                    aimLockCachedTarget = nil
                    return
                end

                local camera = Workspace.CurrentCamera
                if not camera then return end

                local currentCF = camera.CFrame
                local camPos = currentCF.Position
                local currentLook = currentCF.LookVector

                local dirToTarget = targetPos - camPos
                local targetDistFromPlayer = (targetPos - humanoidRootPart.Position).Magnitude

                if targetDistFromPlayer > (VD.AimLockMaxDistance or 130) then
                    aimLockCachedTarget = nil
                    return
                end

                local smoothFactor = math.clamp(VD.AimLockSmoothness or 1, 0.01, 1)
                local newLook = currentLook:Lerp(dirToTarget.Unit, smoothFactor)
                if newLook.Magnitude < 0.001 then return end

                camera.CFrame = CFrame.new(camPos, camPos + newLook.Unit)

                if targetHumanoid and not targetHumanoid.AutoRotate then
                    local lookVec = camera.CFrame.LookVector
                    local flatLook = Vector3.new(lookVec.X, 0, lookVec.Z)
                    if flatLook.Magnitude > 0.001 then
                        local currentAngle = math.atan2(humanoidRootPart.CFrame.LookVector.X, humanoidRootPart.CFrame.LookVector.Z)
                        local targetAngle = math.atan2(flatLook.X, flatLook.Z)
                        local angleDiff = targetAngle - currentAngle
                        angleDiff = ((angleDiff + math.pi) % (2 * math.pi)) - math.pi
                        local newAngle = currentAngle + angleDiff * 0.15
                        local newX = math.sin(newAngle)
                        local newZ = math.cos(newAngle)
                        local targetPosRot = humanoidRootPart.Position + Vector3.new(newX, 0, newZ) * 900
                        humanoidRootPart.CFrame = CFrame.new(humanoidRootPart.Position, Vector3.new(targetPosRot.X, humanoidRootPart.Position.Y, targetPosRot.Z))
                    end
                end
            end)
        end

    local function IYAN_ScanMap()
        local function getTaggedParts(tag)
            local result = {}
            for _, inst in ipairs(CollectionService:GetTagged(tag)) do
                if inst:IsA("BasePart") and inst:IsDescendantOf(Workspace) then
                    table.insert(result, { part = inst, model = inst.Parent })
                elseif inst:IsA("Model") then
                    local part = inst:FindFirstChild("HumanoidRootPart") or inst:FindFirstChild("PrimaryPart") or inst:FindFirstChildWhichIsA("BasePart", true)
                    if part then
                        table.insert(result, { part = part, model = inst })
                    end
                end
            end
            return result
        end

        IYAN_Cache.Generators = getTaggedParts("Generator")
        IYAN_Cache.Hooks = getTaggedParts("Hook")
        IYAN_Cache.Pallets = getTaggedParts("Pallet")
        IYAN_Cache.Gates = getTaggedParts("Gate")
        IYAN_Cache.Windows = getTaggedParts("Window")

        local finish = Workspace:FindFirstChild("Map") and Workspace.Map:FindFirstChild("Finishline")
        if finish then
            local fp = finish:IsA("BasePart") and finish or (finish:IsA("Model") and finish:FindFirstChildWhichIsA("BasePart"))
            if fp then IYAN_Cache.ExitPos = fp.Position end
        end

        local root = Root
        if root and #IYAN_Cache.Hooks > 0 then
            local closest, closestDist = nil, math.huge
            for _, hook in ipairs(IYAN_Cache.Hooks) do
                if hook.part then
                    local d = (hook.part.Position - root.Position).Magnitude
                    if d < closestDist then
                        closestDist = d; closest = hook
                    end
                end
            end
            IYAN_Cache.ClosestHook = closest
        end
    end

    local function solvePitch(p, d, dy)
        d = math.max(d, 0.1)
        local s2 = p.v0 * p.v0
        local root = s2 * s2 - p.g * (p.g * d * d + 2 * dy * s2)
        local outOfRange = false
        if root < 0 then root = 0; outOfRange = true end
        local tanTheta = (s2 - math.sqrt(root)) / (p.g * d)
        local theta = math.atan(tanTheta)
        local t = d / (p.v0 * math.cos(theta))
        return theta, outOfRange, t
    end

    local function getCharacterVelocity(char)
        local root = char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso"))
        if not root or not root:IsA("BasePart") then return Vector3.zero end
        local now = os.clock()
        local last = SpearState.velHistory[char]
        local measured = Vector3.zero
        if last and now - last.t > 0.02 then
            measured = (root.Position - last.pos) / (now - last.t)
            if measured.Magnitude > 150 then measured = last.smooth or Vector3.zero end
        end
        local smooth = last and last.smooth or measured
        smooth = smooth:Lerp(measured, 0.65)
        SpearState.velHistory[char] = { pos = root.Position, t = now, smooth = smooth }
        if smooth.Magnitude < 1 then return Vector3.zero end
        return Vector3.new(smooth.X, 0, smooth.Z)
    end

    local function UpdateSpearAimbot()
        if GetRole() ~= "Killer" then
            SpearState.target = nil
            SpearState.lookVector = nil
            if SpearVisuals.FOVCircleFill then SpearVisuals.FOVCircleFill.Visible = false end

            if SpearVisuals.TrackerCircleOutline then SpearVisuals.TrackerCircleOutline.Visible = false end
            if SpearVisuals.TrackerLine then SpearVisuals.TrackerLine.Visible = false end
            return
        end

        local selectedKiller = LocalPlayer:GetAttribute("SelectedKiller")
        if selectedKiller ~= "Veil" then
            SpearState.target = nil
            SpearState.lookVector = nil
            if SpearVisuals.FOVCircleFill then SpearVisuals.FOVCircleFill.Visible = false end
            if SpearVisuals.FOVCircleOutline then SpearVisuals.FOVCircleOutline.Visible = false end
            if SpearVisuals.TrackerCircleFill then SpearVisuals.TrackerCircleFill.Visible = false end
            if SpearVisuals.TrackerCircleOutline then SpearVisuals.TrackerCircleOutline.Visible = false end
            if SpearVisuals.TrackerLine then SpearVisuals.TrackerLine.Visible = false end
            return
        end

        local cam = Workspace.CurrentCamera
        if not cam then return end
        local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)

        -- FOV Circle visibility
        if VD.ShowFOV and VD.EnableSilentVeil then
            SpearVisuals.FOVCircleOutline.Position = center
            SpearVisuals.FOVCircleOutline.Radius = VD.FovRadius or 150
            SpearVisuals.FOVCircleOutline.Visible = true
            SpearVisuals.FOVCircleFill.Position = center
            SpearVisuals.FOVCircleFill.Radius = VD.FovRadius or 150
            SpearVisuals.FOVCircleFill.Visible = true
        else
            SpearVisuals.FOVCircleOutline.Visible = false
            SpearVisuals.FOVCircleFill.Visible = false
        end

        if not VD.EnableSilentVeil then
            SpearState.target = nil
            SpearState.lookVector = nil
            SpearVisuals.TrackerCircleOutline.Visible = false
            SpearVisuals.TrackerCircleFill.Visible = false
            SpearVisuals.TrackerLine.Visible = false
            return
        end

        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        local nearest = nil
        local bestDist = VD.FovRadius or 150
        local bestStudDist = VD.MaxTargetDistance or 500

        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and IsSurvivor(p) and p.Character then
                local pc = p.Character
                local isDown = pc:GetAttribute("Knocked") == true or pc:GetAttribute("HookProgressDepleting") == true
                if not isDown then
                    local hum = pc:FindFirstChildOfClass("Humanoid")
                    local targetPart = pc:FindFirstChild("UpperTorso") or pc:FindFirstChild("Torso") or pc:FindFirstChild("HumanoidRootPart")
                    if hum and hum.Health > 0 and targetPart then
                        local sp, on = cam:WorldToScreenPoint(targetPart.Position)
                        if on and sp.Z > 0 then
                            local sd = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                            if sd < bestDist then
                                local studDist = (targetPart.Position - hrp.Position).Magnitude
                                if studDist <= bestStudDist then
                                    bestDist = sd
                                    nearest = p
                                end
                            end
                        end
                    end
                end
            end
        end

        if nearest and nearest.Character then
            local tpPart = nearest.Character:FindFirstChild("UpperTorso") or nearest.Character:FindFirstChild("Torso") or nearest.Character:FindFirstChild("HumanoidRootPart")
            if tpPart then
                local tp = tpPart.Position
                local hand = char:FindFirstChild("Right Arm") or char:FindFirstChild("RightHand")
                local origin = (hand and hand:IsA("BasePart")) and hand.Position or hrp.Position
                local dir = tp - origin
                local dist = dir.Magnitude

                if dist > 0.1 and dist <= (VD.MaxTargetDistance or 500) then
                    local isAuraActive = char:GetAttribute("special") == true
                    local prof

                    if isAuraActive then
                        prof = {
                            v0 = VD.AuraSpearSpeed or 165,
                            g = VD.AuraSpearGravity or 96.5,
                            windup = 0.10,
                            latency = 0.04,
                            maxlead = 25,
                            scale = VD.LeadMultiplier or 1.4
                        }
                    else
                        prof = {
                            v0 = VD.CustomSpearSpeed or 165,
                            g = VD.CustomSpearGravity or 103,
                            windup = 0.10,
                            latency = 0.04,
                            maxlead = 45,
                            scale = VD.LeadMultiplier or 1.4
                        }
                    end

                    local aimPoint = tp
                    if VD.AutoPredict then
                        local vel = getCharacterVelocity(nearest.Character)
                        if vel.Magnitude > 0.5 then
                            local h0 = Vector3.new(dir.X, 0, dir.Z)
                            local _, _, tFlight = solvePitch(prof, h0.Magnitude, dir.Y)

                            local ping = 0.08
                            pcall(function() ping = math.clamp(LocalPlayer:GetNetworkPing(), 0, 0.35) end)

                            local delay = tFlight + prof.windup + ping + prof.latency
                            for i = 1, 2 do
                                local lead = vel * delay * prof.scale
                                local maxLead = math.clamp(dist * 0.6, 3, prof.maxlead)
                                if lead.Magnitude > maxLead then lead = lead.Unit * maxLead end
                                aimPoint = tp + lead
                                local ad = aimPoint - origin
                                local ah = Vector3.new(ad.X, 0, ad.Z)
                                local _, _, t2 = solvePitch(prof, math.max(ah.Magnitude, 0.1), ad.Y)
                                delay = t2 + prof.windup + ping + prof.latency
                            end
                        end
                    end

                    local adir = aimPoint - origin
                    local ah = Vector3.new(adir.X, 0, adir.Z)
                    local ahDist = ah.Magnitude
                    local pitch = solvePitch(prof, ahDist, adir.Y)
                    if ahDist > 0.001 then
                        SpearState.lookVector = ah.Unit * math.cos(pitch) + Vector3.new(0, math.sin(pitch), 0)
                    else
                        SpearState.lookVector = adir.Unit
                    end
                    SpearState.target = nearest

                    if VD.ShowTracker then
                        local sp, vis = cam:WorldToScreenPoint(tpPart.Position)
                        if vis and sp.Z > 0 then
                            local screenDist = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                            if screenDist <= (VD.FovRadius or 150) then
                                local bottomCenter = Vector2.new(center.X, cam.ViewportSize.Y)
                                -- Line
                                SpearVisuals.TrackerLine.From = bottomCenter
                                SpearVisuals.TrackerLine.To = Vector2.new(sp.X, sp.Y)
                                SpearVisuals.TrackerLine.Visible = true
                                -- Circle outline + fill
                                local radius = math.clamp(1000 / dist, 10, 40)
                                SpearVisuals.TrackerCircleOutline.Position = Vector2.new(sp.X, sp.Y)
                                SpearVisuals.TrackerCircleOutline.Radius = radius
                                SpearVisuals.TrackerCircleOutline.Visible = true
                                SpearVisuals.TrackerCircleFill.Position = Vector2.new(sp.X, sp.Y)
                                SpearVisuals.TrackerCircleFill.Radius = radius
                                SpearVisuals.TrackerCircleFill.Visible = true
                            else
                                SpearVisuals.TrackerLine.Visible = false
                                SpearVisuals.TrackerCircleOutline.Visible = false
                                SpearVisuals.TrackerCircleFill.Visible = false
                            end
                        else
                            SpearVisuals.TrackerLine.Visible = false
                            SpearVisuals.TrackerCircleOutline.Visible = false
                            SpearVisuals.TrackerCircleFill.Visible = false
                        end
                    else
                        SpearVisuals.TrackerLine.Visible = false
                        SpearVisuals.TrackerCircleOutline.Visible = false
                        SpearVisuals.TrackerCircleFill.Visible = false
                    end
                else
                    SpearState.target = nil
                    SpearState.lookVector = nil
                    SpearVisuals.TrackerLine.Visible = false
                    SpearVisuals.TrackerCircleOutline.Visible = false
                    SpearVisuals.TrackerCircleFill.Visible = false
                end
            else
                SpearState.target = nil
                SpearState.lookVector = nil
                SpearVisuals.TrackerLine.Visible = false
                SpearVisuals.TrackerCircleOutline.Visible = false
                SpearVisuals.TrackerCircleFill.Visible = false
            end
        else
            SpearState.target = nil
            SpearState.lookVector = nil
            SpearVisuals.TrackerLine.Visible = false
            SpearVisuals.TrackerCircleOutline.Visible = false
            SpearVisuals.TrackerCircleFill.Visible = false
        end
    end

    -- Remote hook for silent aim (spearthrow)
    local function SetupSilentVeilHook()
        local mt = getrawmetatable(game)
        if not mt then return end
        local oldNamecall = mt.__namecall

        local newNamecall = function(self, ...)
            local method = getnamecallmethod()
            local args = {...}

            if method == "FireServer" and typeof(self) == "Instance" then
                local selfName = string.lower(self.Name)
                if selfName == "spearthrow" and GetRole(LocalPlayer) == "Killer" and VD.EnableSilentVeil and typeof(SpearState.lookVector) == "Vector3" then
                    if typeof(args[1]) == "Vector3" then
                        args[1] = SpearState.lookVector
                    end
                    return oldNamecall(self, unpack(args))
                end
            end
            return oldNamecall(self, ...)
        end

        setreadonly(mt, false)
        mt.__namecall = newNamecall
        setreadonly(mt, true)
    end

    task.spawn(function()
        pcall(SetupSilentVeilHook)
        RunService.RenderStepped:Connect(function()
            pcall(UpdateSpearAimbot)
        end)
    end)

    LocalPlayer:GetPropertyChangedSignal("Team"):Connect(function()
        if GetRole(LocalPlayer) ~= "Killer" then
            SpearState.target = nil
            SpearState.lookVector = nil
            if SpearVisuals.FOVCircleFill then SpearVisuals.FOVCircleFill.Visible = false end
            if SpearVisuals.FOVCircleOutline then SpearVisuals.FOVCircleOutline.Visible = false end
            if SpearVisuals.TrackerCircleFill then SpearVisuals.TrackerCircleFill.Visible = false end
            if SpearVisuals.TrackerCircleOutline then SpearVisuals.TrackerCircleOutline.Visible = false end
            if SpearVisuals.TrackerLine then SpearVisuals.TrackerLine.Visible = false end
        end
    end)

    -- ============================================================
    -- SYNC ALL FEATURES ON LOAD
    -- ============================================================
    local _isSyncing = false

    function SyncAllFeatures()
        if _isSyncing then return end
        _isSyncing = true

        print("[Gluto] Syncing all features from config...")

        VD_SetAutoSkillcheck(VD.AutoSkillcheck)

        if VD.SURV_AntiKnock then
            if not antiKnockConnection then
                local char = LocalPlayer.Character
                if char then
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    if hum then
                        antiKnockConnection = hum.HealthChanged:Connect(function()
                            hum.Health = 100
                        end)
                    end
                end
            end
        else
            if antiKnockConnection then
                antiKnockConnection:Disconnect()
                antiKnockConnection = nil
            end
        end

        applyNoFog(VD.NoFog)
        applyFullbright(VD.Fullbright)

        if VD.BypassGate then
            pcall(function() setBypassGate(true) end)
        else
            pcall(function() setBypassGate(false) end)
        end

        if VD.SpeedBoost then
            local char = LocalPlayer.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if hum then
                local boost = math.clamp(VD.SpeedBoostPercent or 50, 0, 100) * 2
                local targetSpeed = 16 * (1 + boost / 100)
                pcall(function()
                    hum.WalkSpeed = targetSpeed
                end)
            end
        end

        if VD.FLASH_SilentAim then
            SetFlashlightSilentAim(true)
        else
            SetFlashlightSilentAim(false)
        end

        if VD.KILLER_SilentAimFlask and VD.KILLER_FlaskLaser then
            StartFlaskLaser()
        else
            StopFlaskLaser()
        end

        if VD.KILLER_BypassCooldown then KYS_StartAbyssCooldownBypass() else KYS_StopAbyssCooldownBypass() end
        if VD.KILLER_BypassLeap then KYS_StartHiddenCooldownBypass() else KYS_StopHiddenCooldownBypass() end
        if VD.KILLER_InfFrenzy then KYS_StartJeffCooldownBypass() else KYS_StopJeffCooldownBypass() end
        if VD.KILLER_InfLakeMist or VD.KILLER_InfPursuit then KYS_StartSlasherCooldownBypass() else KYS_StopSlasherCooldownBypass() end

        if VD.AimLockEnabled then
            if type(getgenv()._Gluto_AimLockEnable) == "function" then
                getgenv()._Gluto_AimLockEnable()
            end
        else
            if type(getgenv()._Gluto_AimLockDisable) == "function" then
                getgenv()._Gluto_AimLockDisable()
            end
        end

        if type(getgenv().ESP_RefreshAllPlayer) == "function" then
            getgenv().ESP_RefreshAllPlayer()
        end

        -- Generator Boost sync
        if VD.SURV_GenBoost then
            startGenBoost()
        else
            stopGenBoost()
        end

        print("[Gluto] Sync complete.")
        _isSyncing = false
    end

    task.spawn(function()
        task.wait(1.5)
        pcall(SyncAllFeatures)
    end)

    -- ============================================================

        if UILib and UILib.MakeNotify then

            })
        end
    end)
